Snake Game let snakeInterval = null; function startSnake() { const canvas = document.getElementById('snakeCanvas'); if (!canvas) return; const ctx = canvas.getContext('2d'); if (snakeInterval) clearInterval(snakeInterval); let snake = [{x: 140, y: 100}]; let food = {x: 60, y: 60}; let dx = 20, dy = 0; document.onkeydown = (e) => { if (e.key === 'ArrowUp' && dy === 0) { dx = 0; dy = -20; } if (e.key === 'ArrowDown' && dy === 0) { dx = 0; dy = 20; } if (e.key === 'ArrowLeft' && dx === 0) { dx = -20; dy = 0; } if (e.key === 'ArrowRight' && dx === 0) { dx = 20; dy = 0; } }; snakeInterval = setInterval(() => { let head = {x: snake[0].x + dx, y: snake[0].y + dy}; if (head.x < 0 || head.x >= 280 || head.y < 0 || head.y >= 220) { clearInterval(snakeInterval); alert('Game Over!'); return; } snake.unshift(head); if (head.x === food.x && head.y === food.y) { food = { x: Math.floor(Math.random() * 14) * 20, y: Math.floor(Math.random() * 10) * 20 }; } else { snake.pop(); } ctx.fillStyle = '#1a1a20'; ctx.fillRect(0, 0, 280, 220); ctx.fillStyle = '#f96332'; ctx.fillRect(food.x, food.y, 18, 18); ctx.fillStyle = '#ffffff'; snake.forEach(part => ctx.fillRect(part.x, part.y, 18, 18)); }, 130); }
