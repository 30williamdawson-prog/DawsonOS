// ========================================== // DAWSON OS - CORE SYSTEM SCRIPT // ========================================== document.addEventListener('DOMContentLoaded', () => { initClock(); initWindowManager(); initNotepad(); initCalculator(); initPaintApp(); }); /* --- 1. SYSTEM CLOCK --- */ function initClock() { const clockElement = document.getElementById('taskbar-clock'); if (!clockElement) return; function updateTime() { const now = new Date(); clockElement.innerText = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }); } updateTime(); setInterval(updateTime, 1000); } /* --- 2. WINDOW MANAGEMENT & DRAGGING --- */ let activeZIndex = 100; function initWindowManager() { const windows = document.querySelectorAll('.os-window'); windows.forEach(win => { // Bring window to front on click win.addEventListener('mousedown', () => { activeZIndex++; win.style.zIndex = activeZIndex; }); // Make window draggable via header const header = win.querySelector('.window-header'); if (header) { makeElementDraggable(win, header); } // Close button event const closeBtn = win.querySelector('.btn-close'); if (closeBtn) { closeBtn.addEventListener('click', () => { win.style.display = 'none'; }); } }); } function openWindow(windowId) { const win = document.getElementById(windowId); if (win) { win.style.display = 'block'; activeZIndex++; win.style.zIndex = activeZIndex; } } function makeElementDraggable(elmnt, handle) { let pos1 = 0, pos2 = 0, pos3 = 0, pos4 = 0; handle.onmousedown = dragMouseDown; function dragMouseDown(e) { e = e || window.event; e.preventDefault(); pos3 = e.clientX; pos4 = e.clientY; document.onmouseup = closeDragElement; document.onmousemove = elementDrag; } function elementDrag(e) { e = e || window.event; e.preventDefault(); pos1 = pos3 - e.clientX; pos2 = pos4 - e.clientY; pos3 = e.clientX; pos4 = e.clientY; elmnt.style.top = (elmnt.offsetTop - pos2) + "px"; elmnt.style.left = (elmnt.offsetLeft - pos1) + "px"; } function closeDragElement() { document.onmouseup = null; document.onmousemove = null; } } /* --- 3. NOTEPAD APP (LOCAL STORAGE) --- */ function initNotepad() { const notepadArea = document.getElementById('notepad-textarea'); if (!notepadArea) return; // Load saved note on startup const savedNote = localStorage.getItem('dawson_os_note'); if (savedNote) { notepadArea.value = savedNote; } // Auto-save changes notepadArea.addEventListener('input', () => { localStorage.setItem('dawson_os_note', notepadArea.value); }); } /* --- 4. CALCULATOR APP --- */ let calcDisplayValue = "0"; function initCalculator() { updateCalcDisplay(); } function calcInput(num) { if (calcDisplayValue === "0") { calcDisplayValue = num.toString(); } else { calcDisplayValue += num.toString(); } updateCalcDisplay(); } function calcOperator(op) { calcDisplayValue += ` ${op} `; updateCalcDisplay(); } function calcClear() { calcDisplayValue = "0"; updateCalcDisplay(); } function calcEquals() { try { const result = eval(calcDisplayValue.replace(/×/g, '*').replace(/÷/g, '/')); calcDisplayValue = String(result); } catch (err) { calcDisplayValue = "Error"; } updateCalcDisplay(); } function updateCalcDisplay() { const display = document.getElementById('calc-display'); if (display) { display.innerText = calcDisplayValue; } } /* --- 5. DAWSON PAINT APP --- */ function initPaintApp() { const canvas = document.getElementById('paint-canvas'); if (!canvas) return; const ctx = canvas.getContext('2d'); let isDrawing = false; let currentColor = '#000000'; let brushSize = 5; // Set default background color ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvas.width, canvas.height); // Event handlers canvas.addEventListener('mousedown', startDrawing); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseup', stopDrawing); canvas.addEventListener('mouseout', stopDrawing); const colorPicker = document.getElementById('paint-color'); if (colorPicker) { colorPicker.addEventListener('change', (e) => currentColor = e.target.value); } const sizePicker = document.getElementById('paint-size'); if (sizePicker) { sizePicker.addEventListener('change', (e) => brushSize = e.target.value); } function startDrawing(e) { isDrawing = true; draw(e); } function draw(e) { if (!isDrawing) return; const rect = canvas.getBoundingClientRect(); const x = e.clientX - rect.left; const y = e.clientY - rect.top; ctx.lineWidth = brushSize; ctx.lineCap = 'round'; ctx.strokeStyle = currentColor; ctx.lineTo(x, y); ctx.stroke(); ctx.beginPath(); ctx.moveTo(x, y); } function stopDrawing() { isDrawing = false; ctx.beginPath(); } } function clearCanvas() { const canvas = document.getElementById('paint-canvas'); if (!canvas) return; const ctx = canvas.getContext('2d'); ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvas.width, canvas.height); }
