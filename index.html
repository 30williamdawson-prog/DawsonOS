// ========================================== // DAWSON OS - CORE SYSTEM SCRIPT WITH CHATBOT // ========================================== document.addEventListener('DOMContentLoaded', () => { initClock(); initWindowManager(); initNotepad(); initCalculator(); initPaintApp(); initChatbot(); }); /* --- 1. SYSTEM CLOCK --- */ function initClock() { const clockElement = document.getElementById('taskbar-clock'); if (!clockElement) return; function updateTime() { const now = new Date(); clockElement.innerText = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }); } updateTime(); setInterval(updateTime, 1000); } /* --- 2. WINDOW MANAGEMENT & DRAGGING --- */ let activeZIndex = 100; function initWindowManager() { const windows = document.querySelectorAll('.os-window'); windows.forEach(win => { win.addEventListener('mousedown', () => { activeZIndex++; win.style.zIndex = activeZIndex; }); const header = win.querySelector('.window-header'); if (header) { makeElementDraggable(win, header); } const closeBtn = win.querySelector('.btn-close'); if (closeBtn) { closeBtn.addEventListener('click', () => { win.style.display = 'none'; }); } }); } function openWindow(windowId) { const win = document.getElementById(windowId); if (win) { win.style.display = 'block'; activeZIndex++; win.style.zIndex = activeZIndex; } } function makeElementDraggable(elmnt, handle) { let pos1 = 0, pos2 = 0, pos3 = 0, pos4 = 0; handle.onmousedown = dragMouseDown; function dragMouseDown(e) { e = e || window.event; e.preventDefault(); pos3 = e.clientX; pos4 = e.clientY; document.onmouseup = closeDragElement; document.onmousemove = elementDrag; } function elementDrag(e) { e = e || window.event; e.preventDefault(); pos1 = pos3 - e.clientX; pos2 = pos4 - e.clientY; pos3 = e.clientX; pos4 = e.clientY; elmnt.style.top = (elmnt.offsetTop - pos2) + "px"; elmnt.style.left = (elmnt.offsetLeft - pos1) + "px"; } function closeDragElement() { document.onmouseup = null; document.onmousemove = null; } } /* --- 3. DAWSON AI CHATBOT ENGINE --- */ function initChatbot() { const sendBtn = document.getElementById('chat-send-btn'); const inputField = document.getElementById('chat-input'); if (!sendBtn || !inputField) return; sendBtn.addEventListener('click', handleUserMessage); inputField.addEventListener('keypress', (e) => { if (e.key === 'Enter') handleUserMessage(); }); } function handleUserMessage() { const inputField = document.getElementById('chat-input'); const message = inputField.value.trim(); if (message === '') return; // Display user's message appendChatMessage('User', message); inputField.value = ''; // Simulate AI "thinking" time setTimeout(() => { const aiResponse = generateAIResponse(message); appendChatMessage('Dawson AI', aiResponse); }, 600); } function appendChatMessage(sender, text) { const chatLog = document.getElementById('chat-log'); if (!chatLog) return; const msgDiv = document.createElement('div'); msgDiv.className = sender === 'User' ? 'chat-msg user-msg' : 'chat-msg ai-msg'; msgDiv.style.margin = '8px 0'; msgDiv.style.padding = '8px 12px'; msgDiv.style.borderRadius = '8px'; msgDiv.style.maxWidth = '80%'; if (sender === 'User') { msgDiv.style.backgroundColor = '#2563eb'; msgDiv.style.color = '#ffffff'; msgDiv.style.marginLeft = 'auto'; } else { msgDiv.style.backgroundColor = '#374151'; msgDiv.style.color = '#ffffff'; } msgDiv.innerText = `${sender}: ${text}`; chatLog.appendChild(msgDiv); chatLog.scrollTop = chatLog.scrollHeight; } function generateAIResponse(input) { const lower = input.toLowerCase(); // Pattern matching logic if (lower.includes('hello') || lower.includes('hi') || lower.includes('hey')) { return "Hello! Welcome to Dawson OS. How can I help you today?"; } if (lower.includes('who are you') || lower.includes('name')) { return "I am Dawson AI, the built-in virtual assistant for Dawson OS."; } if (lower.includes('time') || lower.includes('clock')) { return `The current system time is ${new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}.`; } if (lower.includes('calc') || lower.includes('math')) { return "You can open the Calculator app from the taskbar to run mathematical operations."; } if (lower.includes('draw') || lower.includes('paint')) { return "Launch the Dawson Paint app to sketch, change colors, and adjust brush sizes!"; } if (lower.includes('note') || lower.includes('save')) { return "The Notepad app saves your notes directly to your local browser storage automatically."; } if (lower.includes('aphg') || lower.includes('geography')) { return "Remember: Distance decay shows interaction decreasing as distance increases!"; } if (lower.includes('clear')) { const chatLog = document.getElementById('chat-log'); if (chatLog) chatLog.innerHTML = ''; return "Chat log cleared!"; } // Fallback dynamic responses const fallbacks = [ "I'm processing that. Try asking about Dawson OS features like Paint, Notepad, or the Calculator!", "Interesting! How else can I assist your desktop experience?", "I'm continuously learning. Try typing 'time' or asking about 'notes'!", "System running smooth. What app would you like to build next?" ]; return fallbacks[Math.floor(Math.random() * fallbacks.length)]; } /* --- 4. NOTEPAD APP --- */ function initNotepad() { const notepadArea = document.getElementById('notepad-textarea'); if (!notepadArea) return; const savedNote = localStorage.getItem('dawson_os_note'); if (savedNote) notepadArea.value = savedNote; notepadArea.addEventListener('input', () => { localStorage.setItem('dawson_os_note', notepadArea.value); }); } /* --- 5. CALCULATOR APP --- */ let calcDisplayValue = "0"; function initCalculator() { updateCalcDisplay(); } function calcInput(num) { calcDisplayValue = (calcDisplayValue === "0") ? num.toString() : calcDisplayValue + num.toString(); updateCalcDisplay(); } function calcOperator(op) { calcDisplayValue += ` ${op} `; updateCalcDisplay(); } function calcClear() { calcDisplayValue = "0"; updateCalcDisplay(); } function calcEquals() { try { const result = eval(calcDisplayValue.replace(/×/g, '*').replace(/÷/g, '/')); calcDisplayValue = String(result); } catch (err) { calcDisplayValue = "Error"; } updateCalcDisplay(); } function updateCalcDisplay() { const display = document.getElementById('calc-display'); if (display) display.innerText = calcDisplayValue; } /* --- 6. DAWSON PAINT APP --- */ function initPaintApp() { const canvas = document.getElementById('paint-canvas'); if (!canvas) return; const ctx = canvas.getContext('2d'); let isDrawing = false; let currentColor = '#000000'; let brushSize = 5; ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvas.width, canvas.height); canvas.addEventListener('mousedown', (e) => { isDrawing = true; draw(e); }); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseup', () => { isDrawing = false; ctx.beginPath(); }); canvas.addEventListener('mouseout', () => { isDrawing = false; ctx.beginPath(); }); const colorPicker = document.getElementById('paint-color'); if (colorPicker) colorPicker.addEventListener('change', (e) => currentColor = e.target.value); const sizePicker = document.getElementById('paint-size'); if (sizePicker) sizePicker.addEventListener('change', (e) => brushSize = e.target.value); function draw(e) { if (!isDrawing) return; const rect = canvas.getBoundingClientRect(); const x = e.clientX - rect.left; const y = e.clientY - rect.top; ctx.lineWidth = brushSize; ctx.lineCap = 'round'; ctx.strokeStyle = currentColor; ctx.lineTo(x, y); ctx.stroke(); ctx.beginPath(); ctx.moveTo(x, y); } } function clearCanvas() { const canvas = document.getElementById('paint-canvas'); if (!canvas) return; const ctx = canvas.getContext('2d'); ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvas.width, canvas.height); }
